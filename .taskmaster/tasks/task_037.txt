3. **Transparency Processing Logic**:
   - Modify the image processing function:
   ```javascript
   function processImageWithTransparency(img, thresholdValue) {
     // Create a new image with alpha channel support
     let processedImg = createGraphics(img.width, img.height, WEBGL);
     processedImg.pixelDensity(1);
     
     img.loadPixels();
     processedImg.loadPixels();
     
     for (let i = 0; i < img.pixels.length; i += 4) {
       // Calculate gray value (average of RGB)
       const grayValue = (img.pixels[i] + img.pixels[i+1] + img.pixels[i+2]) / 3;
       
       if (transparencyModeEnabled) {
         // Transparency mode logic
         if (grayValue >= 255 - thresholdValue) {
           // White-ish pixels become transparent
           processedImg.pixels[i] = 0;      // R
           processedImg.pixels[i+1] = 0;    // G
           processedImg.pixels[i+2] = 0;    // B
           processedImg.pixels[i+3] = 0;    // A (transparent)
         } else {
           // Black or dark gray pixels remain black
           processedImg.pixels[i] = 0;      // R
           processedImg.pixels[i+1] = 0;    // G
           processedImg.pixels[i+2] = 0;    // B
           processedImg.pixels[i+3] = 255;  // A (opaque)
         }
       } else {
         // Standard mode (non-transparent background)
         const pixelValue = grayValue > thresholdValue ? 255 : 0;
         processedImg.pixels[i] = pixelValue;     // R
         processedImg.pixels[i+1] = pixelValue;   // G
         processedImg.pixels[i+2] = pixelValue;   // B
         processedImg.pixels[i+3] = 255;          // A (opaque)
       }
     }
     
     processedImg.updatePixels();
     return processedImg;
   }
   ```

   - **Solution Implemented**:

       ```javascript
       if (invert) {
         // In inverted mode: original black pixels (now white) should be transparent

         // In normal mode: original white pixels should be transparent
         // original black pixels should remain black
         shouldBeTransparent = avg >= 255 - transparencyThreshold
       }
       ```
