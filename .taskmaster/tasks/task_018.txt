5. **Create reusable test utilities**:
   - Canvas rendering verification:
   ```typescript
   // tests/utils/canvas-helpers.ts
   import { Page, expect } from '@playwright/test';

   export async function verifyCanvasExists(page: Page): Promise<void> {
     const canvas = page.locator('canvas');
     await expect(canvas).toBeVisible();
   }

   export async function verifyCanvasNotEmpty(page: Page): Promise<void> {
     // Get canvas and check pixel data
     const nonEmptyCanvas = await page.evaluate(() => {
       const canvas = document.querySelector('canvas');
       if (!canvas) return false;
       
       const ctx = canvas.getContext('2d');
       const data = ctx.getImageData(0, 0, canvas.width, canvas.height).data;
       
       // Check if all pixels are transparent/empty
       for (let i = 3; i < data.length; i += 4) {
         if (data[i] !== 0) return true;
       }
       return false;
     });
     
     expect(nonEmptyCanvas).toBeTruthy();
   }
   ```

   - Keyboard interaction helpers:
   ```typescript
   // tests/utils/keyboard-helpers.ts
   import { Page } from '@playwright/test';

   export async function pressKeyAndWait(page: Page, key: string, waitTime = 100): Promise<void> {
     await page.keyboard.press(key);
     await page.waitForTimeout(waitTime);
   }

   export async function typeTextAndWait(page: Page, text: string, waitTime = 100): Promise<void> {
     await page.keyboard.type(text);
     await page.waitForTimeout(waitTime);
   }
   ```

   - Mouse interaction helpers:
   ```typescript
   // tests/utils/mouse-helpers.ts
   import { Page } from '@playwright/test';

   export async function clickCanvasAt(page: Page, x: number, y: number): Promise<void> {
     const canvas = page.locator('canvas');
     const box = await canvas.boundingBox();
     
     if (!box) throw new Error('Canvas not found');
     
     await page.mouse.click(box.x + x, box.y + y);
   }

   export async function dragOnCanvas(page: Page, startX: number, startY: number, endX: number, endY: number): Promise<void> {
     const canvas = page.locator('canvas');
     const box = await canvas.boundingBox();
     
     if (!box) throw new Error('Canvas not found');
     
     await page.mouse.move(box.x + startX, box.y + startY);
     await page.mouse.down();
     await page.mouse.move(box.x + endX, box.y + endY);
     await page.mouse.up();
   }
   ```

   - Screenshot comparison utilities:
   ```typescript
   // tests/utils/screenshot-helpers.ts
   import { Page, expect } from '@playwright/test';
   import path from 'path';

   export async function compareScreenshot(page: Page, screenshotName: string, threshold = 0.2): Promise<void> {
     const screenshotPath = path.join('tests', 'screenshots', 'reference', `${screenshotName}.png`);
     
     await expect(page.locator('canvas')).toHaveScreenshot(screenshotName + '.png', {
       maxDiffPixelRatio: threshold,
     });
   }

   export async function captureCanvasScreenshot(page: Page, screenshotName: string): Promise<string> {
     const canvas = page.locator('canvas');
     const screenshotPath = path.join('tests', 'screenshots', 'actual', `${screenshotName}.png`);
     
     await canvas.screenshot({ path: screenshotPath });
     return screenshotPath;
   }
   ```

7. **Create example tests for those-shape-things**:
   ```typescript
   // tests/e2e/those-shape-things.spec.ts
   import { test, expect } from '@playwright/test';
   import { verifyCanvasExists, verifyCanvasNotEmpty } from '../utils/canvas-helpers';
   import { pressKeyAndWait } from '../utils/keyboard-helpers';
   import { clickCanvasAt } from '../utils/mouse-helpers';

   test.beforeEach(async ({ page }) => {
     await page.goto('/apps/those-shape-things/');
     await page.waitForLoadState('networkidle');
   });

   test('should load with canvas visible', async ({ page }) => {
     await verifyCanvasExists(page);
     await verifyCanvasNotEmpty(page);
   });

   test('should toggle background color with "b" key', async ({ page }) => {
     // Get initial background color
     const initialBgColor = await page.evaluate(() => {
       return window.getComputedStyle(document.querySelector('canvas')).backgroundColor;
     });
     
     // Press 'b' to toggle background
     await pressKeyAndWait(page, 'b');
     
     // Get new background color
     const newBgColor = await page.evaluate(() => {
       return window.getComputedStyle(document.querySelector('canvas')).backgroundColor;
     });
     
     // Verify background color changed
     expect(newBgColor).not.toEqual(initialBgColor);
   });

   test('should create new shape on canvas click', async ({ page }) => {
     // Count initial shapes
     const initialShapeCount = await page.evaluate(() => {
       // This assumes there's a shapes array or similar in the app
       // Adjust based on actual implementation
       return window.appInstance?.shapes?.length || 0;
     });
     
     // Click on canvas
     await clickCanvasAt(page, 150, 150);
     
     // Count shapes after click
     const newShapeCount = await page.evaluate(() => {
       return window.appInstance?.shapes?.length || 0;
     });
     
     // Verify a new shape was added
     expect(newShapeCount).toBeGreaterThan(initialShapeCount);
   });
   ```
