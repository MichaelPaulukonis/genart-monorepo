# Details:
This task involves implementing the crop feature for the Monochromifier application according to the PRD:

1. **Mode System Implementation**:
   - Add new application modes: 'VIEW' (existing), 'EDIT', and 'CROP'
   - Implement mode switching via UI controls and keyboard shortcuts
   - Create a mode indicator in the UI to show current active mode
   ```javascript
   const MODES = {
     VIEW: 'VIEW',
     EDIT: 'EDIT',
     CROP: 'CROP'
   };
   let currentMode = MODES.VIEW;
   
   function switchMode(newMode) {
     if (Object.values(MODES).includes(newMode)) {
       currentMode = newMode;
       updateModeIndicator();
     }
   }
   ```

2. **Selection Implementation**:
   - Create a rectangular selection tool that activates in CROP mode
   - Implement click-and-drag functionality to define the crop area
   - Add visual indicators (handles, borders) for the selection area
   - Allow selection adjustment by dragging edges or corners
   ```javascript
   let selection = {
     active: false,
     startX: 0,
     startY: 0,
     endX: 0,
     endY: 0,
     isDragging: false
   };
   
   function mousePressed() {
     if (currentMode === MODES.CROP) {
       selection.active = true;
       selection.startX = mouseX;
       selection.startY = mouseY;
       selection.endX = mouseX;
       selection.endY = mouseY;
       selection.isDragging = true;
     }
   }
   
   function mouseDragged() {
     if (currentMode === MODES.CROP && selection.isDragging) {
       selection.endX = mouseX;
       selection.endY = mouseY;
     }
   }
   ```

3. **Crop Operation Implementation**:
   - Implement the actual crop operation that trims both the source image and paint layers
   - Ensure proper scaling and positioning after crop
   - Update canvas dimensions to match the cropped area
   ```javascript
   function performCrop() {
     if (!selection.active) return;
     
     // Calculate crop dimensions
     const x = min(selection.startX, selection.endX);
     const y = min(selection.startY, selection.endY);
     const w = abs(selection.endX - selection.startX);
     const h = abs(selection.endY - selection.startY);
     
     // Create new buffers for image and paint layers
     const newImageLayer = createGraphics(w, h);
     const newPaintLayer = createGraphics(w, h);
     
     // Copy selected portions to new buffers
     newImageLayer.copy(imageLayer, x, y, w, h, 0, 0, w, h);
     newPaintLayer.copy(paintLayer, x, y, w, h, 0, 0, w, h);
     
     // Replace existing layers with cropped versions
     imageLayer = newImageLayer;
     paintLayer = newPaintLayer;
     
     // Update canvas dimensions
     resizeCanvas(w, h);
     
     // Reset selection
     selection.active = false;
   }
   ```

4. **Undo Integration**:
   - Integrate with the existing undo system (Task 35 approach)
   - Save state before crop operation
   - Allow reverting to pre-crop state
   ```javascript
   function saveCropHistoryState() {
     // Save current state before cropping
     historyStack.push({
       timestamp: Date.now(),
       operation: 'crop',
       imageData: imageLayer.get(),
       paintData: paintLayer.get(),
       canvasWidth: width,
       canvasHeight: height
     });
     
     // Truncate forward history if we're in the middle of the stack
     if (currentHistoryIndex < historyStack.length - 1) {
       historyStack = historyStack.slice(0, currentHistoryIndex + 1);
     }
     
     currentHistoryIndex = historyStack.length - 1;
   }
   ```

5. **UI Controls**:
   - Add buttons for EDIT and CROP modes
   - Implement "Apply Crop" and "Cancel Crop" buttons
   - Add keyboard shortcuts (e.g., 'e' for EDIT, 'c' for CROP, Enter to apply, Esc to cancel)
   ```javascript
   function setupCropUI() {
     // Create mode buttons
     editModeButton = createButton('Edit');
     editModeButton.mousePressed(() => switchMode(MODES.EDIT));
     
     cropModeButton = createButton('Crop');
     cropModeButton.mousePressed(() => switchMode(MODES.CROP));
     
     // Create crop action buttons (initially hidden)
     applyCropButton = createButton('Apply Crop');
     applyCropButton.mousePressed(() => {
       saveCropHistoryState();
       performCrop();
       switchMode(MODES.VIEW);
     });
     
     cancelCropButton = createButton('Cancel');
     cancelCropButton.mousePressed(() => {
       selection.active = false;
       switchMode(MODES.VIEW);
     });
     
     // Update UI visibility based on mode
     updateUIVisibility();
   }
   ```

6. **Visual Feedback**:
   - Implement overlay to darken areas outside the selection
   - Add handles for selection adjustment
   - Provide clear visual indication of current mode
   ```javascript
   function drawCropOverlay() {
     if (currentMode === MODES.CROP && selection.active) {
       // Draw semi-transparent overlay outside selection
       push();
       noStroke();
       fill(0, 0, 0, 100);
       
       // Top
       rect(0, 0, width, min(selection.startY, selection.endY));
       // Bottom
       rect(0, max(selection.startY, selection.endY), width, height - max(selection.startY, selection.endY));
       // Left
       rect(0, min(selection.startY, selection.endY), min(selection.startX, selection.endX), abs(selection.endY - selection.startY));
       // Right
       rect(max(selection.startX, selection.endX), min(selection.startY, selection.endY), 
            width - max(selection.startX, selection.endX), abs(selection.endY - selection.startY));
       
       // Draw selection border
       stroke(255);
       noFill();
       rect(min(selection.startX, selection.endX), min(selection.startY, selection.endY), 
            abs(selection.endX - selection.startX), abs(selection.endY - selection.startY));
       
       // Draw handles at corners
       drawSelectionHandles();
       pop();
     }
   }
   ```
