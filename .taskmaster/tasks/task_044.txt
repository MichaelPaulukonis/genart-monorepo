   const canvasSizePresets = {
     square: { width: 800, height: 800, label: "Square (1:1)" },

2. **Implement Canvas Resizing Logic**:
   - Create a function to handle canvas resizing while preserving content:
   ```javascript
   function resizeCanvas(newWidth, newHeight) {
     // Create temporary buffer to store current canvas content
     const tempBuffer = createGraphics(canvas.width, canvas.height);
     tempBuffer.image(canvas, 0, 0);
     
     // Resize the main canvas
     resizeCanvas(newWidth, newHeight);
     
     // Center the previous content on new canvas
     const xOffset = (width - tempBuffer.width) / 2;
     const yOffset = (height - tempBuffer.height) / 2;
     image(tempBuffer, xOffset, yOffset);
     
     // Clean up
     tempBuffer.remove();
     
     // Update any size-dependent variables
     updateBrushScalingFactors();
   }
   ```
