# Details:
Implement save functionality for the those-shape-things app:

1. Add event listeners for keyboard shortcuts:
   ```javascript
   function keyPressed() {
     // Check for 's' key or Cmd+S (metaKey is Cmd on Mac)
     if (key === 's' || (keyCode === 83 && metaKey)) {
       saveComposition();
       return false; // Prevent default browser save dialog
     }
   }
   ```

2. Create the save function that generates a timestamped filename and saves the canvas:
   ```javascript
   function saveComposition() {
     // Generate timestamp in format YYYYMMDD-HHMMSS
     let now = new Date();
     let timestamp = now.getFullYear() +
                    String(now.getMonth() + 1).padStart(2, '0') +
                    String(now.getDate()).padStart(2, '0') + '-' +
                    String(now.getHours()).padStart(2, '0') +
                    String(now.getMinutes()).padStart(2, '0') +
                    String(now.getSeconds()).padStart(2, '0');
     
     // Create filename with timestamp
     let filename = 'those-shape-things-' + timestamp + '.png';
     
     // Use p5.js saveCanvas() function to save the current canvas as PNG
     saveCanvas(canvas, filename, 'png');
     
     // Optional: Provide visual feedback that the image was saved
     displaySaveNotification();
   }
   ```

3. Add visual feedback for the user when an image is saved:
   ```javascript
   function displaySaveNotification() {
     // Create a temporary notification element
     let notification = document.createElement('div');
     notification.className = 'save-notification';
     notification.textContent = 'Composition saved!';
     document.body.appendChild(notification);
     
     // Remove the notification after 2 seconds
     setTimeout(() => {
       notification.classList.add('fade-out');
       setTimeout(() => {
         document.body.removeChild(notification);
       }, 500);
     }, 2000);
   }
   ```

   }
