# Task ID: 65
# Title: Implement Build-Time Image List Generation for Duo-Chrome App
# Status: done
# Dependencies: 31, 34
# Priority: medium
# Description: Develop a system that automatically scans public/images/*.png files during build time and generates a JavaScript module with an array of image filenames that can be imported by the duo-chrome application, using Vite build hooks to update whenever PNG files are added or removed.
# Details:
This task involves implementing a build-time image list generation system specifically for the duo-chrome application:

1. **Create a Vite Plugin**:
   ```javascript
   // vite-image-list-plugin.js
   import fs from 'fs/promises';
   import path from 'path';
   
   export default function imageListPlugin() {
     return {
       name: 'image-list-generator',
       buildStart: async () => {
         // This will run at the start of each build
         await generateImageList();
       },
       // Add a file watcher for the images directory
       configureServer(server) {
         server.watcher.add('public/images/**/*.png');
         server.watcher.on('add', (path) => {
           if (path.includes('public/images') && path.endsWith('.png')) {
             generateImageList();
           }
         });
         server.watcher.on('unlink', (path) => {
           if (path.includes('public/images') && path.endsWith('.png')) {
             generateImageList();
           }
         });
       }
     };
   }
   ```

2. **Implement Image List Generation Function**:
   ```javascript
   async function generateImageList() {
     const imagesDir = path.resolve('public/images');
     try {
       // Get all PNG files in the directory
       const files = await fs.readdir(imagesDir);
       const pngFiles = files.filter(file => file.toLowerCase().endsWith('.png'));
       
       // Create the output as a JavaScript module
       const outputContent = `// Auto-generated file - DO NOT EDIT
// Generated on ${new Date().toISOString()}

/**
 * Array of available images for duo-chrome app
 * @type {string[]}
 */
const imageList = ${JSON.stringify(pngFiles, null, 2)};

export default imageList;
`;
       
       // Write the output file to the src directory so it can be imported
       await fs.writeFile(
         path.resolve('src/apps/duo-chrome/image-list.js'),
         outputContent
       );
       
       console.log(`Generated image list with ${pngFiles.length} images`);
     } catch (error) {
       console.error('Error generating image list:', error);
     }
   }
   ```

3. **Configure Vite Plugin in Build Config**:
   ```javascript
   // vite.config.js
   import { defineConfig } from 'vite';
   import imageListPlugin from './vite-image-list-plugin';
   
   export default defineConfig({
     plugins: [
       // ... other plugins
       imageListPlugin()
     ]
   });
   ```

4. **Update Duo-Chrome App to Use Generated Image List**:
   ```javascript
   // In the duo-chrome app file where images are loaded
   import imageList from './image-list.js';
   
   // Use the image list in the application
   function getRandomImage() {
     const randomIndex = Math.floor(Math.random() * imageList.length);
     return `/images/${imageList[randomIndex]}`;
   }
   
   // Update UI to display available images
   function populateImageSelector(selectElement) {
     imageList.forEach(image => {
       const option = document.createElement('option');
       option.value = image;
       option.textContent = image.replace('.png', '');
       selectElement.appendChild(option);
     });
   }
   ```

5. **Handle Edge Cases**:
   - Add error handling for empty directories
   - Ensure the output directory exists before writing
   - Implement logging for debugging purposes
   - Add support for nested directories if needed in the future
